2. StudentDAO.java (Add Update & Delete Methods)

public int updateStudent(Student s)
throws Exception
{
    PreparedStatement ps =
    con.prepareStatement(
    "update student set sname=?, course=? where sid=?");

    ps.setString(1,s.getSname());
    ps.setString(2,s.getCourse());
    ps.setInt(3,s.getSid());

    return ps.executeUpdate();
}

public int deleteStudent(int sid)
throws Exception
{
    PreparedStatement ps =
    con.prepareStatement(
    "delete from student where sid=?");

    ps.setInt(1,sid);

    return ps.executeUpdate();
}

3: MenuServlet.java


package com.cdac.servlet;

import java.io.IOException;

import javax.servlet.annotation.WebServlet;
import javax.servlet.http.*;

@WebServlet("/menu")
public class MenuServlet extends HttpServlet
{
    protected void doPost(
            HttpServletRequest request,
            HttpServletResponse response)
            throws IOException
    {
        int choice =
        Integer.parseInt(
        request.getParameter("choice"));

        switch(choice)
        {
            case 1:

                response.sendRedirect(
                "addstudent.html");
                break;

            case 2:

                response.sendRedirect(
                "view");
                break;

            case 3:

                response.sendRedirect(
                "updatestudent.html");
                break;

            case 4:

                response.sendRedirect(
                "deletestudent.html");
                break;

            case 5:

                response.sendRedirect(
                "logout");
                break;

            default:

                response.getWriter().println(
                "Invalid Choice");
        }
    }
}

4: UpdateStudentServlet.java:


package com.cdac.servlet;

import java.io.IOException;

import javax.servlet.annotation.WebServlet;
import javax.servlet.http.*;

import com.cdac.dao.StudentDAO;
import com.cdac.model.Student;

@WebServlet("/update")
public class UpdateStudentServlet
extends HttpServlet
{
    protected void doPost(
            HttpServletRequest request,
            HttpServletResponse response)
            throws IOException
    {
        try
        {
            Student s =
            new Student();

            s.setSid(
            Integer.parseInt(
            request.getParameter("sid")));

            s.setSname(
            request.getParameter("sname"));

            s.setCourse(
            request.getParameter("course"));

            StudentDAO dao =
            new StudentDAO();

            int n =
            dao.updateStudent(s);

            if(n > 0)
            {
                response.getWriter().println(
                "Student Updated Successfully");
            }
        }
        catch(Exception e)
        {
            e.printStackTrace();
        }
    }
}

package com.cdac.servlet;

import java.io.IOException;

import javax.servlet.annotation.WebServlet;
import javax.servlet.http.*;

@WebServlet("/delete")
public class DeleteStudentServlet
extends HttpServlet
{
    protected void doPost(
            HttpServletRequest request,
            HttpServletResponse response)
            throws IOException
    {
        try
        {
            int sid =

            int n =

            {
                response.getWriter().println(

        }
        catch(Exception e)
        {
            e.printStackTrace();
        }
    }
}
